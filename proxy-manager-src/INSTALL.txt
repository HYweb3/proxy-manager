快速安装（3步完成）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

步骤1: 解压文件
  tar -xzf proxy-manager-v1.0.tar.gz
  cd proxy-manager

步骤2: 运行安装
  sudo bash quick_install.sh

步骤3: 访问管理面板
  http://your-server-ip:5080

系统要求
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CPU架构:  x86_64, aarch64, armv7l
内  存:   最低 512MB，推荐 1GB+
权  限:   Root 或 sudo

端口配置（从500开始，避免冲突）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

VLESS:       端口 500 (TLS加密)
Trojan:      端口 501 (TLS加密)
VMess:       端口 502 (TLS加密)
Shadowsocks: 端口 503 (无加密)
Web管理:     端口 5080

安全特性
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

python3 /etc/proxy-manager/proxy_manager.py list

# 重启服务
sudo systemctl restart xray proxy-web

# 查看服务状态
sudo systemctl status xray proxy-web

# 查看访问日志
